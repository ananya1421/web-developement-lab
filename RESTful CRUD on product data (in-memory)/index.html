const express=require('express'); const app=express(); app.use(express.json()); let products=[ {id:1,name:'Pen',price:10,description:'Blue ink pen'} ]; app.get('/products',(req,res)=>res.json(products)); app.get('/products/:id',(req,res)=>{ const p=products.find(x=>x.id===+req.params.id); p?res.json(p):res.status(404).send('Product not found'); }); app.post('/products',(req,res)=>{ const p={id:Date.now(),...req.body}; products.push(p); res.status(201).json(p); }); app.put('/products/:id',(req,res)=>{ const i=products.findIndex(x=>x.id===+req.params.id); if(i<0)return res.status(404).send('Not found'); products[i]={...products[i],...req.body,id:+req.params.id}; res.json(products[i]); }); app.delete('/products/:id',(req,res)=>{ products=products.filter(x=>x.id!==+req.params.id); res.send('Product deleted'); }); app.listen(3000,()=>console.log('Server started'));
